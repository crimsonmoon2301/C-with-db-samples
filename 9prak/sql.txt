CREATE TABLE Projekts (
    ProjektsId SERIAL PRIMARY KEY,
    Nosaukums VARCHAR(100) NOT NULL
);

CREATE TABLE Ipasnieks (
    IpasnieksId SERIAL PRIMARY KEY,
    Vards VARCHAR(100),
    Uzvards VARCHAR(100)
);

CREATE TABLE Maja (
    MajaId SERIAL PRIMARY KEY,
    Adrese TEXT NOT NULL,
    Platiba INT,
    IpasnieksId INT REFERENCES Ipasnieks(IpasnieksId),
    ProjektsId INT REFERENCES Projekts(ProjektsId)
);
